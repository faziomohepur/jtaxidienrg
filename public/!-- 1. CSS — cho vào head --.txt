<!-- 2. HTML — đặt nơi muốn hiển thị player -->
<div class="music-player">
  <h2>🎵 Music Player</h2>
  <div class="now-playing" id="nowPlaying">Đang tải danh sách...</div>
  <ul class="playlist" id="playlist"><li>⏳ Đang tải...</li></ul>
  <audio id="audio" controls></audio>
  <button class="start-btn" id="startBtn">▶ Bắt đầu phát nhạc</button>
</div>

<!-- 3. JS — cho vào cuối <body> -->
<script>
  // ⚠️ Thay bằng URL Worker của ông
  const API_URL = "https://music-player.wtf1.workers.dev";

  let songs = [], currentIndex = -1, userInteracted = false;
  const playlist = document.getElementById('playlist');
  const audio = document.getElementById('audio');
  const nowPlaying = document.getElementById('nowPlaying');
  const startBtn = document.getElementById('startBtn');

  async function loadSongs() {
    try {
      const res = await fetch(API_URL + '/api/songs');
      const data = await res.json();
      songs = data.songs || [];
      if (songs.length === 0) {
        playlist.innerHTML = '<li>Chưa có file MP3 nào</li>';
        return;
      }
      renderPlaylist();
      nowPlaying.textContent = 'Chọn một bài hát để phát';
    } catch (err) {
      playlist.innerHTML = '<li>❌ Lỗi tải danh sách</li>';
    }
  }

  function renderPlaylist() {
    playlist.innerHTML = '';
    songs.forEach((song, i) => {
      const li = document.createElement('li');
      li.innerHTML = '<span>' + song.title + '</span><span class="song-size">' + formatSize(song.size) + '</span>';
      li.addEventListener('click', () => playSong(i));
      playlist.appendChild(li);
    });
  }

  function playSong(index) {
    currentIndex = index;
    const song = songs[index];
    // Gọi qua API Worker — không đổi URL
    audio.src = API_URL + song.url;
    nowPlaying.textContent = 'Đang phát: ' + song.title;
    document.querySelectorAll('.playlist li').forEach((li, i) => li.classList.toggle('active', i === index));
    audio.play().catch(() => {
      if (!userInteracted) nowPlaying.textContent = '👆 Bấm play để nghe: ' + song.title;
    });
  }

  function formatSize(bytes) {
    if (bytes < 1024) return bytes + ' B';
    if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' KB';
    return (bytes / 1048576).toFixed(1) + ' MB';
  }

  audio.addEventListener('ended', () => {
    if (currentIndex < songs.length - 1) playSong(currentIndex + 1);
  });

  startBtn.addEventListener('click', () => {
    userInteracted = true;
    startBtn.classList.add('hidden');
    if (songs.length > 0) playSong(0);
  });

  loadSongs();
</script>
